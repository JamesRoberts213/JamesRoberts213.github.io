How the Firmware Renders This Frame

  • All 18 Holes Embedded: Uses the exact 18-hole WGS84 coordinate array (`courseHoles[18]`) from your firmware, defaulting to Hole 12 (Par 3) at startup[cite: 7].
  • Dynamic Vector Navigation: Computes great-circle distance (`calculateDistanceYards`) and initial forward azimuth (`calculateTargetBearing`) to rotate the compass needle and output `TURN LEFT`, `ON TARGET` ($\pm 8^\circ$), or `TURN RIGHT`[cite: 7].
  • Automated Club Caddie: Evaluates `getClubRecommendation(yards)` across 10 distance brackets—from `DRIVER / 3-WOOD` ($>220\text{ yds}$) down to `LOB WEDGE / PUTT` ($\le 60\text{ yds}$)[cite: 7].
  • Flicker-Free Dirty-Rectangle Redraws: Only clears and repaints the dynamic numeric bounding boxes (`fillRect` over `COLOR_BG`) rather than clearing the entire $240 \times 320$ display buffer[cite: 7].

// ---------- MATH HELPERS ----------
double degToRad(double degrees) { return degrees * PI / 180.0; }
double radToDeg(double radians) { return radians * 180.0 / PI; }

double normaliseDegrees(double degrees) {
  while (degrees < 0) degrees += 360.0;
  while (degrees >= 360.0) degrees -= 360.0;
  return degrees;
}

double calculateDistanceYards(double lat1, double lon1, double lat2, double lon2) {
  const double earthRadiusMetres = 6371000.0;

  double dLat = degToRad(lat2 - lat1);
  double dLon = degToRad(lon2 - lon1);

  double a = sin(dLat / 2) * sin(dLat / 2) +
             cos(degToRad(lat1)) * cos(degToRad(lat2)) *
             sin(dLon / 2) * sin(dLon / 2);

  double c = 2 * atan2(sqrt(a), sqrt(1 - a));
  double distanceMetres = earthRadiusMetres * c;

  return distanceMetres / 0.9144;
}

double calculateTargetBearing(double lat1, double lon1, double lat2, double lon2) {
  double longitudeDifference = degToRad(lon2 - lon1);

  double y = sin(longitudeDifference) * cos(degToRad(lat2));
  double x = cos(degToRad(lat1)) * sin(degToRad(lat2)) -
             sin(degToRad(lat1)) * cos(degToRad(lat2)) *
             cos(longitudeDifference);

  return normaliseDegrees(radToDeg(atan2(y, x)));
}

const char *turnDirection() {

int getGPSBars() {
  if (!gps.satellites.isValid()) return 0;
  int satellites = gps.satellites.value();

  if (satellites <= 3) return 1;
  if (satellites <= 6) return 2;
  if (satellites <= 9) return 3;
  return 4;
}

// ---------- NAVIGATION & DRAWING ----------
void updateNavigation() {
  if (!gps.location.isValid() || !compassOK) {
    navigationValid = false;
    return;
  }

  double currentLat = gps.location.lat();
  double currentLon = gps.location.lng();

  double targetLat = courseHoles[currentHole - 1].lat;
  double targetLon = courseHoles[currentHole - 1].lon;

  distanceToTargetYards = calculateDistanceYards(currentLat, currentLon, targetLat, targetLon);
  targetBearing         = calculateTargetBearing(currentLat, currentLon, targetLat, targetLon);

  turnAngle = targetBearing - heading;

  if (turnAngle > 180)  turnAngle -= 360;
  if (turnAngle < -180) turnAngle += 360;

  navigationValid = true;
}

// ---------- SETUP & LOOP ----------
void setup() {
  Serial.begin(9600);

  pinMode(BTN_HOLE_UP, INPUT_PULLUP);
  pinMode(BTN_HOLE_DOWN, INPUT_PULLUP);
  pinMode(BTN_REFRESH, INPUT_PULLUP);

  pinMode(TFT_BACKLIGHT, OUTPUT);
  digitalWrite(TFT_BACKLIGHT, HIGH);

  tft.init(240, 320);
  tft.setRotation(0);
  tft.fillScreen(COLOR_BG);

  gpsSerial.begin(9600);
  Wire.begin();

  if (compass.begin()) {
    compass.setMode(QMC5883P_MODE_CONTINUOUS);
    compass.setODR(QMC5883P_ODR_50HZ);
    compass.setRange(QMC5883P_RANGE_8G);
    compassOK = true;
  }

  drawStaticLayout();
  updateEverything();
}

void loop() {
  // 1. Process incoming GPS data continuously
  while (gpsSerial.available()) {
    gps.encode(gpsSerial.read());
  }

  // 2. Read calibrated and aligned compass sensor
  if (compassOK && compass.isDataReady()) {
    int16_t rawX, rawY, rawZ;
    if (compass.getRawMagnetic(&rawX, &rawY, &rawZ)) {
      // Apply hard-iron offset calibration
      float x = (float)rawX - X_OFFSET;
      float y = (float)rawY - Y_OFFSET;
      float z = (float)rawZ - Z_OFFSET;

      // Calculate base angle
      float tempHeading = atan2(y, x) * 180.0 / PI;

      // Apply +90 deg alignment shift & declination
      tempHeading += 90.0;
      tempHeading += MAGNETIC_DECLINATION_DEG;

      // Flip East and West across N-S axis
      tempHeading = 360.0 - tempHeading;

      // Keep within 0 to 359 degrees
      heading = normaliseDegrees(tempHeading);
    }
  }

  // 3. Read Buttons
  bool currentHoleUp   = digitalRead(BTN_HOLE_UP);
  bool currentHoleDown = digitalRead(BTN_HOLE_DOWN);
  bool currentRefresh  = digitalRead(BTN_REFRESH);

  if (lastHoleUpState == HIGH && currentHoleUp == LOW) {
    currentHole++;
    if (currentHole > 18) currentHole = 1;

    drawStaticLayout();
    showRefreshIndicator();
    updateEverything();
    delay(40);
  }

  if (lastHoleDownState == HIGH && currentHoleDown == LOW) {
    currentHole--;
    if (currentHole < 1) currentHole = 18;

    drawStaticLayout();
    showRefreshIndicator();
    updateEverything();
    delay(40);
  }

  if (lastRefreshState == HIGH && currentRefresh == LOW) {
    showRefreshIndicator();
    updateEverything();
    delay(40);
  }

  lastHoleUpState   = currentHoleUp;
  lastHoleDownState = currentHoleDown;
  lastRefreshState  = currentRefresh;
}

Circuit Architecture & Layout Strategy

  • Multi-Bus Sensor Integration: Routed dedicated traces for 4-wire hardware SPI (`CS D10`, `DC D7`, `RST D8`), I2C magnetometer telemetry (`QMC5883P` at $50\text{ Hz}$), and `SoftwareSerial` UART (`D2/D3` at $9600\text{ baud}$)[cite: 7].
  • Hard-Iron Magnetic Isolation: Kept high-current backlight (`D9`) and power traces away from the QMC5883P sensor package, paired with firmware hard-iron offset calibration ($X = -1282.5, Y = -283.0, Z = +1157.5$)[cite: 7].
  • Mixed-Voltage Level Shifting: Integrated resistive voltage dividers to safely step 5V ATmega328P SPI signals down to 3.3V logic for the ST7789V display controller.
  • Power Distribution & BMS: Routed wide power traces ($20\text{ mil}$) for the TP4056 Li-Ion charger, physical power switch, and decoupling capacitors ($100\text{ nF}$ ceramic + $10\text{ }\mu\text{F}$ bulk) adjacent to IC power pins.

Key Engineering Takeaways

Magnetometer Hard-Iron Calibration

Eliminated heading distortion caused by nearby steel battery casings and PCB traces by applying 3-axis hard-iron offsets ($X = -1282.5, Y = -283.0, Z = +1157.5$) and $+90^\circ$ frame alignment[cite: 7].

3.3V / 5V Logic Level Shifting

Protected the 3.3V ST7789V display controller from 5V Arduino logic levels using resistive voltage dividers on the SPI clock and data lines to ensure long-term signal integrity.

Design for Assembly (DFA) in FDM

Learned how wall thickness around bosses affects melt flow when installing heat-set brass inserts at $210^\circ\text{C}$, preventing exterior surface bulging on thin PLA walls.

Flicker-Free Partial SPI Redraws

Avoided full-screen clearing on the $240 \times 320$ display by repainting only the dynamic yardage, bearing, and compass dial bounding boxes each update cycle[cite: 7].

Non-Blocking NMEA Stream Parsing
